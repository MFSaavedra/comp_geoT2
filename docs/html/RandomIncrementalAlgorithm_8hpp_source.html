102 bool moverTrianguloAlInicio() {
103 const std::size_t n = p.size();
104 std::size_t j = 1;
105 while (j < n && mismoPunto(p[j], p[0])) {
106 ++j;
107 }
108 std::size_t k = j + 1;
109 while (k < n && orientacion(p[0], p[j], p[k]) == 0) {
110 ++k;
111 }
112 if (k >= n) {
113 return false;
114 }
115 std::swap(p[1], p[j]);
116 std::swap(p[2], p[k]);
117 if (orientacion(p[0], p[1], p[2]) < 0) {
118 std::swap(p[1], p[2]);
119 }
120 return true;
121 }

142 Poligono<T> construir() {
143 const int n = static_cast<int>(p.size());
144 sig.assign(n, NINGUNA);
145 ant.assign(n, NINGUNA);
146 arista.assign(n, NINGUNA);
147 cubeta.assign(n, {});
148
149 sig[0] = 1, sig[1] = 2, sig[2] = 0;
150 ant[0] = 2, ant[1] = 0, ant[2] = 1;
151 sumaX = W(p[0].getX()) + W(p[1].getX()) + W(p[2].getX());
152 sumaY = W(p[0].getY()) + W(p[1].getY()) + W(p[2].getY());

154 for (int q = 3; q < n; ++q) {
155 asignarEnTriangulo(q);

159 for (int q = 3; q < n; ++q) {

161 insertar(q);
162 ultimo = q;

189 void insertar(int q) {
190 // 1. Cadena de aristas visibles, desde arista[q] hacia ambos lados. Una
191 // arista cuya recta pasa por q tambien se elimina: q queda en su
192 // prolongacion, y el vertice que comparten deja de ser extremo. Los
193 // topes (ant[l] != fin, r != l) solo actuan con aritmetica inexacta.
194 const int inicio = arista[q];
195 const int fin = sig[inicio];
196 int l = inicio;
197 while (ant[l] != fin && orientacion(p[ant[l]], p[l], p[q]) <= 0) {
198 l = ant[l];
199 }
200 int r = fin;
201 while (r != l && orientacion(p[r], p[sig[r]], p[q]) <= 0) {
202 r = sig[r];
203 }
204
205 // 2. Sacar las cubetas de las aristas eliminadas, de l hasta r.
206 eliminadas.clear();
207 for (int v = l; v != r; v = sig[v]) {
208 eliminadas.push_back(std::move(cubeta[v]));
209 cubeta[v].clear();
210 }
211
212 // 3. Reemplazar la cadena l .. r por l -> q -> r.
213 sig[l] = q, ant[q] = l;
214 sig[q] = r, ant[r] = q;
215 arista[q] = NINGUNA;
216
217 // 4. Cada punto de esas cubetas cruza ahora l -> q o q -> r: lo decide de
218 // que lado del rayo C -> q esta.
219 for (const std::vector<int> &vieja : eliminadas) {
220 for (int f : vieja) {
221 if (f != q) {
222 ++reasignaciones;
223 asignar(f, ladoDelRayo(q, f) < 0 ? l : q);
224 }
225 }
226 }
227 }

235 void asignar(int f, int v) {
236 if (orientacion(p[v], p[sig[v]], p[f]) < 0) {
237 arista[f] = v;
238 cubeta[v].push_back(f);
239 } else {
240 arista[f] = NINGUNA;
241 }
242 }

253 int ladoDelRayo(int v, int f) const {
254 const Punto<W> origen;
255 const Punto<W> cv(3 * W(p[v].getX()) - sumaX, 3 * W(p[v].getY()) - sumaY);
256 const Punto<W> cf(3 * W(p[f].getX()) - sumaX, 3 * W(p[f].getY()) - sumaY);
257 return orientacion(origen, cv, cf);
258 }

265 Poligono<T> recorrer(int desde) const {
266 std::vector<Punto<T>> vertices;
267 int v = desde;
268 do {
269 vertices.push_back(p[v]);
270 v = sig[v];
271 } while (v != desde);
272 Poligono<T> cerradura(vertices);
273 cerradura.normalizar();
274 return cerradura;
275 }
