27template <typename T> class IncrementalAlgorithm : public EstrategiaCerradura<T> {
28public:
34 Poligono<T> apply(const std::vector<Punto<T>> &nube) override {
35 std::vector<Punto<T>> p(nube);
36 std::sort(p.begin(), p.end()); // por x, y a igual x por y
37 p.erase(std::unique(p.begin(), p.end(), mismoPunto<T>), p.end());
38 if (p.size() < 3) {
39 return Poligono<T>(p); // vacio, un punto o un segmento, ya ordenados
40 }
41
42 // Upper hull, de izquierda a derecha. Por arriba, avanzar hacia la
43 // derecha es avanzar en CW: solo se aceptan vueltas a la derecha, y un
44 // punto colineal (orientacion 0) tambien se saca.
45 std::vector<Punto<T>> cadena;
46 for (const Punto<T> &q : p) {
47 while (cadena.size() >= 2 &&
48 orientacion(cadena[cadena.size() - 2], cadena.back(), q) >= 0) {
49 cadena.pop_back();
50 }
51 cadena.push_back(q);
52 }
53
54 // Lower hull, devolviendose de derecha a izquierda desde el extremo
55 // derecho, que ya esta en la cadena y no se puede sacar.
56 const std::size_t base = cadena.size();
57 for (auto q = p.rbegin() + 1; q != p.rend(); ++q) {
58 while (cadena.size() > base &&
59 orientacion(cadena[cadena.size() - 2], cadena.back(), *q) >= 0) {
60 cadena.pop_back();
61 }
62 cadena.push_back(*q);
63 }
64 cadena.pop_back(); // el extremo izquierdo quedo repetido al final
65
66 Poligono<T> cerradura(cadena); // en CW
67 cerradura.normalizar();
68 return cerradura;
69 }
70
72 std::string nombre() const override { return "incremental"; }
