123public:
128 explicit GeneradorCircunferencia(double porcentaje) : porcentaje(porcentaje) {}
129
137 std::vector<Punto<T>> generar(std::size_t n,
138 std::mt19937_64 &rng) const override {
139 const double R = this->RADIO, pi = this->PI;
140 const std::size_t k = this->cantidad(porcentaje, n, 3);
141 std::uniform_real_distribution<double> u01(0.0, 1.0);
142 std::vector<Punto<T>> puntos;
143 puntos.reserve(n);
144
145 const double desfase = 2 * pi * u01(rng);
146 for (std::size_t i = 0; i < k; ++i) {
147 const double angulo = desfase + 2 * pi * i / k;
148 puntos.push_back(this->crear(R * std::cos(angulo), R * std::sin(angulo)));
149 }
150
151 const double radioInterior = 0.999 * R * std::cos(pi / k);
152 for (std::size_t i = k; i < n; ++i) {
153 const double rho = radioInterior * std::sqrt(u01(rng)); // uniforme en area
154 const double angulo = 2 * pi * u01(rng);
155 puntos.push_back(this->crear(rho * std::cos(angulo), rho * std::sin(angulo)));
156 }
157
158 std::shuffle(puntos.begin(), puntos.end(), rng);
159 return puntos;
160 }
161
163 std::string nombre() const override { return "circunferencia"; }
164
165private:
166 double porcentaje;

182public:
188 GeneradorCuadrado(double porcentaje, double angulo)
189 : porcentaje(porcentaje), angulo(angulo) {}
190
197 std::vector<Punto<T>> generar(std::size_t n,
198 std::mt19937_64 &rng) const override {
199 const double R = this->RADIO;
200 const std::size_t k = this->cantidad(porcentaje, n, 4);
201 std::uniform_real_distribution<double> lado(-R, R);
202 std::uniform_real_distribution<double> dentro(-0.999 * R, 0.999 * R);
203 std::uniform_int_distribution<int> cualLado(0, 3);
204 std::vector<Punto<T>> puntos;
205 puntos.reserve(n);
206
207 const double esquinas[4][2] = {{-R, -R}, {R, -R}, {R, R}, {-R, R}};
208 for (std::size_t i = 0; i < n; ++i) {
209 double x, y;
210 if (i < 4 && i < k) {
211 x = esquinas[i][0];
212 y = esquinas[i][1];
213 } else if (i < k) {
214 const double t = lado(rng);
215 switch (cualLado(rng)) {
216 case 0: x = t, y = -R; break;
217 case 1: x = R, y = t; break;
218 case 2: x = t, y = R; break;
219 default: x = -R, y = t; break;
220 }
221 } else {
222 x = dentro(rng);
223 y = dentro(rng);
224 }
225 puntos.push_back(rotado(x, y));
226 }
227
228 std::shuffle(puntos.begin(), puntos.end(), rng);
229 return puntos;
230 }
231
233 std::string nombre() const override { return "cuadrado"; }
234
235private:
236 double porcentaje;
237 double angulo;
238
246 Punto<T> rotado(double x, double y) const {
247 const double c = std::cos(angulo), s = std::sin(angulo);
248 return this->crear(c * x - s * y, s * x + c * y);
249 }
