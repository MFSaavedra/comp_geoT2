156 std::vector<Punto<T>> generar(std::size_t n,
157 std::mt19937_64 &rng) const override {
158 const double R = this->RADIO, pi = this->PI;
159 const std::size_t k = this->cantidad(porcentaje, n, 3);
160 std::uniform_real_distribution<double> u01(0.0, 1.0);
161 std::vector<Punto<T>> puntos;
162 puntos.reserve(n);
163
164 const double desfase = 2 * pi * u01(rng);
165 for (std::size_t i = 0; i < k; ++i) {
166 const double angulo = desfase + 2 * pi * i / k;
167 puntos.push_back(this->crear(R * std::cos(angulo), R * std::sin(angulo)));
168 }
169
170 const double radioInterior = 0.999 * R * std::cos(pi / k);
171 for (std::size_t i = k; i < n; ++i) {
172 const double rho = radioInterior * std::sqrt(u01(rng)); // uniforme en area
173 const double angulo = 2 * pi * u01(rng);
174 puntos.push_back(this->crear(rho * std::cos(angulo), rho * std::sin(angulo)));
175 }
176
177 std::shuffle(puntos.begin(), puntos.end(), rng);
178 return puntos;
179 }

216 std::vector<Punto<T>> generar(std::size_t n,
217 std::mt19937_64 &rng) const override {
218 const double R = this->RADIO;
219 const std::size_t k = this->cantidad(porcentaje, n, 4);
220 std::uniform_real_distribution<double> lado(-R, R);
221 std::uniform_real_distribution<double> dentro(-0.999 * R, 0.999 * R);
222 std::uniform_int_distribution<int> cualLado(0, 3);
223 std::vector<Punto<T>> puntos;
224 puntos.reserve(n);
225
226 const double esquinas[4][2] = {{-R, -R}, {R, -R}, {R, R}, {-R, R}};
227 for (std::size_t i = 0; i < n; ++i) {
228 double x, y;
229 if (i < 4 && i < k) {
230 x = esquinas[i][0];
231 y = esquinas[i][1];
232 } else if (i < k) {
233 const double t = lado(rng);
234 switch (cualLado(rng)) {
235 case 0: x = t, y = -R; break;
236 case 1: x = R, y = t; break;
237 case 2: x = t, y = R; break;
238 default: x = -R, y = t; break;
239 }
240 } else {
241 x = dentro(rng);
242 y = dentro(rng);
243 }
244 puntos.push_back(rotado(x, y));
245 }
246
247 std::shuffle(puntos.begin(), puntos.end(), rng);
248 return puntos;
249 }
250
252 std::string nombre() const override { return "cuadrado"; }
253
254private:
255 double porcentaje;
256 double angulo;
257
265 Punto<T> rotado(double x, double y) const {
266 const double c = std::cos(angulo), s = std::sin(angulo);
267 return this->crear(c * x - s * y, s * x + c * y);
268 }
