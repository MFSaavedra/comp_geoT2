144 std::vector<Punto<T>> generar(std::size_t n,
145 std::mt19937_64 &rng) const override {
146 const double R = this->RADIO, pi = this->PI;
147 const std::size_t k = this->cantidad(porcentaje, n, 3);
148 std::uniform_real_distribution<double> u01(0.0, 1.0);
149 std::vector<Punto<T>> puntos;
150 puntos.reserve(n);
151
152 const double desfase = 2 * pi * u01(rng);
153 for (std::size_t i = 0; i < k; ++i) {
154 const double angulo = desfase + 2 * pi * i / k;
155 puntos.push_back(this->crear(R * std::cos(angulo), R * std::sin(angulo)));
156 }
157
158 const double radioInterior = 0.999 * R * std::cos(pi / k);
159 for (std::size_t i = k; i < n; ++i) {
160 const double rho = radioInterior * std::sqrt(u01(rng)); // uniforme en area
161 const double angulo = 2 * pi * u01(rng);
162 puntos.push_back(this->crear(rho * std::cos(angulo), rho * std::sin(angulo)));
163 }
164
165 std::shuffle(puntos.begin(), puntos.end(), rng);
166 return puntos;
167 }
168
170 std::string nombre() const override { return "circunferencia"; }
171
172private:
173 double porcentaje;

204 std::vector<Punto<T>> generar(std::size_t n,
205 std::mt19937_64 &rng) const override {
206 const double R = this->RADIO;
207 const std::size_t k = this->cantidad(porcentaje, n, 4);
208 std::uniform_real_distribution<double> lado(-R, R);
209 std::uniform_real_distribution<double> dentro(-0.999 * R, 0.999 * R);
210 std::uniform_int_distribution<int> cualLado(0, 3);
211 std::vector<Punto<T>> puntos;
212 puntos.reserve(n);
213
214 const double esquinas[4][2] = {{-R, -R}, {R, -R}, {R, R}, {-R, R}};
215 for (std::size_t i = 0; i < n; ++i) {
216 double x, y;
217 if (i < 4 && i < k) {
218 x = esquinas[i][0];
219 y = esquinas[i][1];
220 } else if (i < k) {
221 const double t = lado(rng);
222 switch (cualLado(rng)) {
223 case 0: x = t, y = -R; break;
224 case 1: x = R, y = t; break;
225 case 2: x = t, y = R; break;
226 default: x = -R, y = t; break;
227 }
228 } else {
229 x = dentro(rng);
230 y = dentro(rng);
231 }
232 puntos.push_back(rotado(x, y));
233 }
234
235 std::shuffle(puntos.begin(), puntos.end(), rng);
236 return puntos;
237 }
238
240 std::string nombre() const override { return "cuadrado"; }
241
242private:
243 double porcentaje;
244 double angulo;
245
253 Punto<T> rotado(double x, double y) const {
254 const double c = std::cos(angulo), s = std::sin(angulo);
255 return this->crear(c * x - s * y, s * x + c * y);
256 }
